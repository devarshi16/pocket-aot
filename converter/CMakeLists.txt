cmake_minimum_required(VERSION 3.13)
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
 set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}" CACHE PATH "Pico SDK")
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
 message(FATAL_ERROR "Set PICO_SDK_PATH to a Raspberry Pi Pico SDK checkout")
endif()
if(NOT PICO_TOOLCHAIN_PATH AND DEFINED ENV{PICO_TOOLCHAIN_PATH})
 set(PICO_TOOLCHAIN_PATH "$ENV{PICO_TOOLCHAIN_PATH}" CACHE PATH "Arm toolchain")
endif()
set(PICO_BOARD pico CACHE STRING "pico (RP2040) or pico2 / pico2_w (RP2350)")
set(PICO_NO_PICOTOOL 1 CACHE BOOL "Local UF2 converter")
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(pocket_aot C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
if(NOT CART_DIR)
 message(FATAL_ERROR "Set CART_DIR to a generated cartridge directory")
endif()
add_executable(pocket_aot runtime/pico.c runtime/p8.c "${CART_DIR}/game.c" "${CART_DIR}/assets.c")
target_include_directories(pocket_aot PRIVATE runtime)
if(PICO_RP2350)
 pico_set_linker_script(pocket_aot "${CMAKE_CURRENT_LIST_DIR}/memmap_pocket_rp2350.ld")
 # RP2350 has twice the RAM: give the Lua heap pools most of it.
 target_compile_definitions(pocket_aot PRIVATE P8_LARGE_POOLS PICO_STACK_SIZE=0xa000)
else()
 pico_set_linker_script(pocket_aot "${CMAKE_CURRENT_LIST_DIR}/memmap_pocket.ld")
 target_compile_definitions(pocket_aot PRIVATE PICO_STACK_SIZE=0x4000)
endif()
target_compile_options(pocket_aot PRIVATE -O2 -fstack-usage -Wall -Wextra -Wno-unused-parameter -Wno-unused-variable -Wno-misleading-indentation)
target_compile_definitions(pocket_aot PRIVATE PICO_CORE1_STACK_SIZE=0x1000 PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0 PICO_STDIO_USB_DEFAULT_CRLF=0)
target_link_libraries(pocket_aot pico_stdlib pico_multicore hardware_spi hardware_pwm m)
pico_enable_stdio_usb(pocket_aot 1)
pico_enable_stdio_uart(pocket_aot 0)
pico_add_extra_outputs(pocket_aot)
